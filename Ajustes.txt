$resp = Invoke-RestMethod -Method Post -Uri "https://accounts.zoho.com/oauth/v2/token" -Body @{
  refresh_token = $REFRESH
  client_id     = $CLIENT_ID
  client_secret = $CLIENT_SECRET
  grant_type    = "refresh_token"
}

# O script só deve continuar se um access_token foi recebido
